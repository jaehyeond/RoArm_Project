"""Supplementary contact-band figure (NEW artifact, overwrites nothing).
Both D341/D324 figures of the main run use a 0-8 mm / 0-50 mm y-scale, in which
the +/-0.1 mm regime that carries the whole result is invisible.  This zooms it."""
import csv, math
import numpy as np, matplotlib
matplotlib.use("Agg"); import matplotlib.pyplot as plt
OUT="/home/cgxr/Documents/Robotics/RoArm_Project/claudedocs/runtime_logs/grasp_track/g0b_d420/"
rows=list(csv.DictReader(open(OUT+"t3d_perstep_curves.csv")))
C={1:"#1f77b4",2:"#2ca02c",3:"#d62728"}
def col(leg,k): return np.array([float(r[k]) for r in rows if int(r["leg"])==leg])
def st(leg): return np.array([int(r["physics_step"]) for r in rows if int(r["leg"])==leg])
fig,ax=plt.subplots(2,2,figsize=(17,9.2)); fig.subplots_adjust(hspace=0.32,wspace=0.18)
for a,key,ttl in ((ax[0][0],"clr_fixed_mm","FIXED jaw (link5 / part_029) — contact band"),
                  (ax[0][1],"clr_moving_mm","MOVING jaw (gripper_link / part_063) — contact band")):
    for L in (1,2,3):
        a.plot(st(L),col(L,key+"_MA"),"-",lw=1.5,color=C[L],label=f"leg {L}  M-A")
        a.plot(st(L),col(L,key+"_MB"),":",lw=1.1,color=C[L],alpha=.85,label=f"leg {L}  M-B")
    a.axhline(0,color="k",lw=1.2); a.set_ylim(-0.12,0.12); a.set_xlim(384,536)
    a.axhspan(-0.0005,0.0005,color="k",alpha=.10)
    a.set_title(ttl+"   (y = ±0.12 mm)",fontsize=11); a.grid(alpha=.3)
    a.set_ylabel("signed clearance [mm]"); a.legend(fontsize=7,ncol=3)
for nm,L,s0,s1,q in (("P-a",3,388,389,"clr_fixed_mm"),("P-b",1,490,491,"clr_fixed_mm"),
                     ("P-c",3,500,501,"clr_moving_mm")):
    a=ax[0][0] if q=="clr_fixed_mm" else ax[0][1]
    a.axvline(s0,color=C[L],ls="--",lw=1.0,alpha=.75)
    a.annotate(f"{nm} leg{L}\npredicted {s0}→{s1}",xy=(s0,0.09),fontsize=8,color=C[L])
a=ax[1][0]; s=st(3)
mk=(s>=515)
a.plot(s[mk],col(3,"clr_fixed_mm_MA")[mk],"-o",ms=3,lw=1.6,color="#d62728",label="fixed  M-A")
a.plot(s[mk],col(3,"clr_fixed_mm_MB")[mk],":",lw=1.2,color="#d62728",label="fixed  M-B")
a.plot(s[mk],col(3,"clr_moving_mm_MA")[mk],"-o",ms=3,lw=1.6,color="#9467bd",label="moving M-A")
a.plot(s[mk],col(3,"clr_moving_mm_MB")[mk],":",lw=1.2,color="#9467bd",label="moving M-B")
a.axhline(0,color="k",lw=1.2); a.axvline(519.5,color="grey",ls="-.",lw=1.0)
a.annotate("hold → LIFT",xy=(520,6.0),fontsize=9,color="grey")
a.set_title("★ leg 3 tail 515–534 — the window the 44th range (386–517) truncated\n"
            "fixed jaw lets go, moving jaw never does, object still does not follow",fontsize=11)
a.set_xlabel("physics step"); a.set_ylabel("signed clearance [mm]"); a.grid(alpha=.3); a.legend(fontsize=8)
a=ax[1][1]
a.plot(s[mk],col(3,"clr_moving_mm_MA")[mk],"-o",ms=4,lw=1.8,color="#9467bd",label="moving M-A")
a.plot(s[mk],col(3,"clr_moving_mm_MB")[mk],":s",ms=3,lw=1.2,color="#9467bd",label="moving M-B")
a.axhline(0,color="k",lw=1.2); a.set_ylim(-0.45,0.06); a.axvline(519.5,color="grey",ls="-.",lw=1.0)
a.set_title("same window, moving jaw only (y = −0.45…+0.06 mm)\n"
            "penetration 9–380 µm — compare hull sampling pitch 500 µm",fontsize=11)
a.set_xlabel("physics step"); a.grid(alpha=.3); a.legend(fontsize=8)
fig.suptitle("g0b_d420 t3d_perstep — SUPPLEMENTARY contact-band zoom "
             "(new artifact, overwrites nothing; authority = t3d_perstep_results.json)",fontsize=12)
fig.savefig(OUT+"t3d_perstep_contactband.png",dpi=115,bbox_inches="tight")
print("wrote t3d_perstep_contactband.png")
