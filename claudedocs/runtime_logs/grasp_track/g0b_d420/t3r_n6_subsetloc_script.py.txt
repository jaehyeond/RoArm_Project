def sha256(p):
    h = hashlib.sha256()
    with open(p, "rb") as f:
        for c in iter(lambda: f.read(1 << 20), b""):
            h.update(c)
    return h.hexdigest()


def load_binary_stl(path):
    """VERBATIM from the Gate-0 audit — raw triangle soup, (n,3,3), file units."""
    data = Path(path).read_bytes()
    n = struct.unpack("<I", data[80:84])[0]
    body = np.frombuffer(data[84:84 + n * 50], dtype=np.uint8).reshape(n, 50)
    return body[:, 12:48].copy().view("<f4").reshape(n, 3, 3).astype(np.float64)


def sample_triangles(tris_m, spacing_m):
    """VERBATIM from the Gate-0 audit."""
    out_ = [tris_m.reshape(-1, 3)]
    e1 = np.linalg.norm(tris_m[:, 1] - tris_m[:, 0], axis=1)
    e2 = np.linalg.norm(tris_m[:, 2] - tris_m[:, 0], axis=1)
    e3 = np.linalg.norm(tris_m[:, 2] - tris_m[:, 1], axis=1)
    n_per = np.maximum(1, np.ceil(np.maximum(e1, np.maximum(e2, e3)) / spacing_m).astype(np.int64))
    for n in np.unique(n_per):
        idx = np.flatnonzero(n_per == n)
        ii, jj = np.meshgrid(np.arange(n + 1), np.arange(n + 1), indexing="ij")
        mask = (ii + jj) <= n
        u = (ii[mask] / n)[None, :, None]
        v = (jj[mask] / n)[None, :, None]
        a = tris_m[idx, 0][:, None, :]
        b = tris_m[idx, 1][:, None, :]
        c = tris_m[idx, 2][:, None, :]
        out_.append((a + u * (b - a) + v * (c - a)).reshape(-1, 3))
    return np.vstack(out_)


def depth_r_mm(pts_m):
    return (pts_m[:, 2] - TCP_LOCAL[2]) * 1000.0, np.hypot(pts_m[:, 0], pts_m[:, 1]) * 1000.0


def gate0_metrics(pts_m, label):
    depth, r = depth_r_mm(pts_m)
    az = np.degrees(np.arctan2(pts_m[:, 1], pts_m[:, 0]))
    win = r <= FINGER_WINDOW_R_MM
    fp = r <= FOOTPRINT_R_MM
    wall = (r >= WALL_ANNULUS_MM[0]) & (r <= WALL_ANNULUS_MM[1]) & win
    rim = (depth >= RIM_BAND_MM[0]) & (depth <= RIM_BAND_MM[1]) & win
    m = {"label": label, "n_pts": int(len(pts_m)), "n_in_finger_window": int(win.sum())}
    if win.any():
        j = int(np.argmax(np.where(win, depth, -np.inf)))
        m.update(l_vis_mm=float(depth[j]), r_at_l_vis_mm=float(r[j]),
                 azimuth_at_l_vis_deg=float(az[j]),
                 peak_xyz_mm=[float(v * 1000.0) for v in pts_m[j]])
    else:
        m["l_vis_mm"] = None
    m["max_depth_in_footprint_mm"] = float(depth[fp].max()) if fp.any() else None
    m["max_depth_in_wall_annulus_mm"] = float(depth[wall].max()) if wall.any() else None
    m["n_pts_in_rim_band"] = int(rim.sum())
    m["min_r_mm_in_rim_band"] = float(r[rim].min()) if rim.any() else None
    lv = m.get("l_vis_mm")
    m["tri_state"] = ("ABORT" if lv is None else "PASS" if lv > L_MIN_MM + INDET_HALF_MM
                      else "FAIL" if lv < L_MIN_MM - INDET_HALF_MM else "INDETERMINATE")
    return m


def dstats(d):
    d = np.asarray(d, float)
    return {"n": int(d.size), "median": float(np.median(d)), "mean": float(d.mean()),
            "p95": float(np.percentile(d, 95)), "p99": float(np.percentile(d, 99)),
            "max": float(d.max()), "n_gt_0p1": int((d > 0.1).sum()),
            "n_gt_0p5": int((d > 0.5).sum()), "n_gt_1": int((d > 1.0).sum()),
            "n_gt_2": int((d > 2.0).sum()), "frac_gt_1": float((d > 1.0).mean())}


def hull_diameter(V):
    h = ConvexHull(V)
    P = V[h.vertices]
    return float(np.linalg.norm(P[:, None, :] - P[None, :, :], axis=-1).max()), int(len(h.vertices))


def surf_dist(mesh, pts):
    _, d, _ = trimesh.proximity.closest_point(mesh, pts)
    return np.abs(np.asarray(d, float))


def load_mesh(p):
    m = trimesh.load(str(p), process=False)
    m.merge_vertices()
    return m
